"""FastAPI dependencies for authentication and role-based authorization."""
from fastapi import Depends
from fastapi.security import HTTPAuthorizationCredentials, HTTPBearer

from app.core.errors import AppError
from app.core.security import decode_access_token
from app.db.mongo import get_db, to_object_id

bearer_scheme = HTTPBearer(auto_error=False)


def get_current_user(
    creds: HTTPAuthorizationCredentials | None = Depends(bearer_scheme),
    db=Depends(get_db),
) -> dict:
    """Loads the user from the DB on every request, so role changes and
    deactivation take effect immediately (the role is NOT trusted from the token)."""
    if creds is None:
        raise AppError(401, "NOT_AUTHENTICATED", "Please log in.")
    payload = decode_access_token(creds.credentials)
    if not payload or "sub" not in payload:
        raise AppError(401, "INVALID_TOKEN", "Your session is invalid or has expired.")
    try:
        user = db.users.find_one({"_id": to_object_id(payload["sub"])})
    except AppError:
        user = None
    if not user:
        raise AppError(401, "INVALID_TOKEN", "Your session is invalid or has expired.")
    if not user.get("is_active", True):
        raise AppError(403, "ACCOUNT_DISABLED", "This account has been disabled.")
    return user


def require_role(*roles: str):
    def checker(user: dict = Depends(get_current_user)) -> dict:
        if user["role"] not in roles:
            raise AppError(403, "FORBIDDEN", "You do not have permission to do this.")
        return user

    return checker


def require_approved_seller(user: dict = Depends(require_role("seller")), db=Depends(get_db)) -> dict:
    """Seller whose profile an admin has approved (needed to publish listings)."""
    profile = db.seller_profiles.find_one({"user_id": user["_id"]})
    if not profile or profile.get("status") != "approved":
        raise AppError(403, "SELLER_NOT_APPROVED", "Your seller account is not approved yet.")
    return user

"""One error format for the whole API: {"error": {"code", "message", "details"}}."""
from fastapi import FastAPI, Request
from fastapi.exceptions import RequestValidationError
from fastapi.responses import JSONResponse
from starlette.exceptions import HTTPException as StarletteHTTPException


class AppError(Exception):
    def __init__(self, status_code: int, code: str, message: str, details=None):
        self.status_code = status_code
        self.code = code
        self.message = message
        self.details = details


def _body(code: str, message: str, details=None) -> dict:
    return {"error": {"code": code, "message": message, "details": details}}


def register_error_handlers(app: FastAPI) -> None:
    @app.exception_handler(AppError)
    async def app_error_handler(_: Request, exc: AppError):
        return JSONResponse(status_code=exc.status_code, content=_body(exc.code, exc.message, exc.details))

    @app.exception_handler(RequestValidationError)
    async def validation_handler(_: Request, exc: RequestValidationError):
        details = [
            {"field": ".".join(str(p) for p in e["loc"] if p != "body"), "message": e["msg"]}
            for e in exc.errors()
        ]
        return JSONResponse(status_code=422, content=_body("VALIDATION_ERROR", "Invalid request data.", details))

    @app.exception_handler(StarletteHTTPException)
    async def http_handler(_: Request, exc: StarletteHTTPException):
        return JSONResponse(status_code=exc.status_code, content=_body("HTTP_ERROR", str(exc.detail)))

"""Password hashing (bcrypt), JWT tokens, and Fernet encryption for NID data."""
from datetime import datetime, timedelta, timezone

import bcrypt
import jwt
from cryptography.fernet import Fernet

from app.core.config import get_settings


def hash_password(password: str) -> str:
    return bcrypt.hashpw(password.encode("utf-8"), bcrypt.gensalt()).decode("utf-8")


def verify_password(password: str, password_hash: str) -> bool:
    try:
        return bcrypt.checkpw(password.encode("utf-8"), password_hash.encode("utf-8"))
    except ValueError:
        return False


def create_access_token(user_id: str, role: str) -> str:
    s = get_settings()
    now = datetime.now(timezone.utc)
    payload = {"sub": user_id, "role": role, "iat": now, "exp": now + timedelta(minutes=s.jwt_expire_minutes)}
    return jwt.encode(payload, s.jwt_secret, algorithm=s.jwt_algorithm)


def decode_access_token(token: str) -> dict | None:
    """Returns the payload, or None if the token is invalid or expired."""
    s = get_settings()
    try:
        return jwt.decode(token, s.jwt_secret, algorithms=[s.jwt_algorithm])
    except jwt.PyJWTError:
        return None


def _fernet() -> Fernet:
    return Fernet(get_settings().fernet_key.encode("utf-8"))


def encrypt_text(plain: str) -> str:
    return _fernet().encrypt(plain.encode("utf-8")).decode("utf-8")


def decrypt_text(token: str) -> str:
    return _fernet().decrypt(token.encode("utf-8")).decode("utf-8")

"""Pydantic schemas for users and authentication."""
import re
from typing import Literal, Optional

from pydantic import BaseModel, EmailStr, Field, field_validator, model_validator

PHONE_RE = re.compile(r"^\+?[0-9\- ]{7,20}$")


class RegisterRequest(BaseModel):
    full_name: str = Field(min_length=2, max_length=100)
    email: EmailStr
    phone: str
    password: str = Field(min_length=8, max_length=72)  # bcrypt limit is 72 bytes
    role: Literal["user", "seller"] = "user"  # "admin" is rejected here on purpose
    business_name: Optional[str] = Field(default=None, max_length=120)

    @field_validator("phone")
    @classmethod
    def check_phone(cls, v: str) -> str:
        if not PHONE_RE.match(v):
            raise ValueError("Enter a valid phone number.")
        return v

    @field_validator("password")
    @classmethod
    def check_password_bytes(cls, v: str) -> str:
        if len(v.encode("utf-8")) > 72:
            raise ValueError("Password is too long (max 72 bytes).")
        return v

    @model_validator(mode="after")
    def seller_needs_business_name(self):
        if self.role == "seller" and not (self.business_name or "").strip():
            raise ValueError("business_name is required for sellers.")
        return self


class LoginRequest(BaseModel):
    email: EmailStr
    password: str = Field(min_length=1, max_length=72)


class UserPublic(BaseModel):

from typing import Literal, Optional

from pydantic import BaseModel, Field, field_validator, model_validator


class SellerProductCreate(BaseModel):
    """Used for both POST (create) and PUT (full update). seller_id/moderation/source are never read from the client."""
    title: str = Field(min_length=2, max_length=150)
    description: str = Field(default="", max_length=2000)
    brand: Optional[str] = Field(default=None, max_length=80)
    category_slug: str = Field(min_length=1, max_length=60)
    subcategory_slug: str = Field(min_length=1, max_length=60)
    price_paisa: int = Field(gt=0, le=1_000_000_000)           # BDT x 100
    stock_qty: int = Field(ge=0, le=1_000_000)
    images: list[str] = Field(default_factory=list, max_length=8)  # URLs only, no uploads
    quantity: Optional[float] = Field(default=None, gt=0)      # in g / ml / pcs / pack
    unit: Optional[Literal["g", "ml", "pcs", "pack"]] = None
    attributes: dict[str, str] = Field(default_factory=dict)
    status: Literal["active", "inactive"] = "active"

    @field_validator("title")
    @classmethod
    def strip_title(cls, v: str) -> str:
        v = v.strip()
        if len(v) < 2:
            raise ValueError("Title is too short.")
        return v

    @field_validator("brand")
    @classmethod
    def clean_brand(cls, v):
        v = (v or "").strip()
        return v or None

    @field_validator("images")
    @classmethod
    def check_images(cls, v: list[str]) -> list[str]:
        for url in v:
            if not url.lower().startswith(("http://", "https://")) or len(url) > 500:
                raise ValueError("Images must be http(s) URLs (max 500 characters).")
        return v

    @field_validator("attributes")
    @classmethod
    def check_attributes(cls, v: dict[str, str]) -> dict[str, str]:
        if len(v) > 20 or any(len(k) > 100 or len(x) > 100 for k, x in v.items()):
            raise ValueError("Max 20 attributes, each key/value up to 100 characters.")
        return v

    @model_validator(mode="after")
    def quantity_and_unit_together(self):
        if (self.quantity is None) != (self.unit is None):
            raise ValueError("Give both quantity and unit, or neither.")
        return self

"""FairShare seller products: seller CRUD, public browse, and effective_price (reads M3's promotions)."""
import re

from app.core.errors import AppError
from app.db.mongo import to_object_id
from app.scrapers.matcher import exact_group_key
from app.scrapers.normalizer import parse_quantity_unit
from app.services.category_service import validate_category_pair
from app.services.match_service import sync_group
from app.services.time_utils import aware, iso, utcnow


def _rx(text: str) -> dict:
    return {"$regex": re.escape(text.strip()), "$options": "i"}


def availability_of(doc: dict) -> str:
    return "in_stock" if doc.get("stock_qty", 0) > 0 else "out_of_stock"


def effective_price(db, doc: dict, now=None):
    """(price_paisa to pay, discount_percent or None). The server decides; clients never send prices.
    Promotion contract (docs/DATABASE.md): seller_id, product_ids[], discount_percent 1-90, starts_at, ends_at, is_active."""
    now = now or utcnow()
    price = doc["price_paisa"]
    best = 0.0
    promos = db.promotions.find({
        "seller_id": {"$in": [doc["seller_id"], str(doc["seller_id"])]},
        "is_active": True,
        "product_ids": {"$in": [doc["_id"], str(doc["_id"])]},
    })
    for promo in promos:
        start, end = aware(promo.get("starts_at")), aware(promo.get("ends_at"))
        if (start and start > now) or (end and end < now):
            continue
        best = max(best, float(promo.get("discount_percent") or 0))
    if best <= 0:
        return price, None
    best = min(best, 90.0)
    return max(1, round(price * (100 - best) / 100)), best


def business_names(db, seller_ids) -> dict:
    ids = list(set(seller_ids))
    return {p["user_id"]: p.get("business_name") for p in db.seller_profiles.find({"user_id": {"$in": ids}})}


def public_filter(db) -> dict:
    """Products shoppers may see: active, not hidden by moderation, and the seller is approved."""
    approved = [p["user_id"] for p in db.seller_profiles.find({"status": "approved"})]
    return {"status": "active", "moderation_status": "visible", "seller_id": {"$in": approved}}


def serialize(db, doc: dict, names=None, owner: bool = False, now=None) -> dict:
    price, pct = effective_price(db, doc, now)
    if names is None:
        names = business_names(db, [doc["seller_id"]])
    out = {
        "id": str(doc["_id"]), "seller_id": str(doc["seller_id"]), "seller_name": names.get(doc["seller_id"]),
        "title": doc["title"], "description": doc.get("description", ""), "brand": doc.get("brand"),
        "category_slug": doc["category_slug"], "subcategory_slug": doc["subcategory_slug"],
        "price_paisa": doc["price_paisa"], "effective_price_paisa": price, "discount_percent": pct,
        "currency": doc.get("currency", "BDT"), "availability": availability_of(doc),
        "images": doc.get("images", []), "quantity": doc.get("quantity"), "unit": doc.get("unit"),
        "attributes": doc.get("attributes", {}), "source": "fairshare",
    }
    if owner:
        out.update({"stock_qty": doc["stock_qty"], "status": doc["status"],
                    "moderation_status": doc.get("moderation_status", "visible"),
                    "created_at": iso(doc.get("created_at")), "updated_at": iso(doc.get("updated_at"))})
    return out


def _page(items, total, page, page_size) -> dict:
    return {"items": items, "total": total, "page": page, "page_size": page_size}


def _fields(db, data) -> dict:
    validate_category_pair(db, data.category_slug, data.subcategory_slug)
    quantity, unit = data.quantity, data.unit
    if quantity is None:
        parsed = parse_quantity_unit(data.title)   # e.g. "Rice 5 kg" -> (5000.0, "g")
        if parsed:
            quantity, unit = parsed
    return {
        "title": data.title, "description": data.description, "brand": data.brand,
        "category_slug": data.category_slug, "subcategory_slug": data.subcategory_slug,
        "price_paisa": data.price_paisa, "currency": "BDT", "stock_qty": data.stock_qty,
        "status": data.status, "images": data.images, "quantity": quantity, "unit": unit,
        "attributes": data.attributes,
        "group_key": exact_group_key({"name": data.title, "brand": data.brand, "quantity": quantity, "unit": unit}),
    }


# ---------- seller side (owner only) ----------

def create_product(db, seller: dict, data) -> dict:
    now = utcnow()
    doc = {**_fields(db, data), "seller_id": seller["_id"], "moderation_status": "visible",
           "source": "fairshare", "created_at": now, "updated_at": now}
    doc["_id"] = db.seller_products.insert_one(doc).inserted_id
    sync_group(db, doc["group_key"])
    return serialize(db, doc, owner=True)


def get_owned(db, seller: dict, product_id: str) -> dict:
    doc = db.seller_products.find_one({"_id": to_object_id(product_id, "Product")})
    if not doc or doc.get("status") == "removed":
        raise AppError(404, "NOT_FOUND", "Product not found.")
    if doc["seller_id"] != seller["_id"]:
        raise AppError(403, "FORBIDDEN", "You can only manage your own products.")
    return doc


def list_own(db, seller: dict, page: int, page_size: int) -> dict:
    flt = {"seller_id": seller["_id"], "status": {"$ne": "removed"}}
    total = db.seller_products.count_documents(flt)
    cur = db.seller_products.find(flt).sort([("created_at", -1), ("_id", 1)]).skip((page - 1) * page_size).limit(page_size)
    names = business_names(db, [seller["_id"]])
    return _page([serialize(db, d, names, owner=True) for d in cur], total, page, page_size)


def update_product(db, seller: dict, product_id: str, data) -> dict:
    old = get_owned(db, seller, product_id)
    fields = {**_fields(db, data), "updated_at": utcnow()}
    db.seller_products.update_one({"_id": old["_id"]}, {"$set": fields})
    for key in {old.get("group_key"), fields["group_key"]}:
        sync_group(db, key)
    return serialize(db, db.seller_products.find_one({"_id": old["_id"]}), owner=True)


def remove_product(db, seller: dict, product_id: str) -> None:
    """Soft delete (status 'removed'), so future order history can still refer to the product."""
    old = get_owned(db, seller, product_id)
    db.seller_products.update_one({"_id": old["_id"]}, {"$set": {"status": "removed", "updated_at": utcnow()}})
    sync_group(db, old.get("group_key"))


def set_inventory(db, seller: dict, product_id: str, stock_qty: int) -> dict:
    old = get_owned(db, seller, product_id)
    db.seller_products.update_one({"_id": old["_id"]}, {"$set": {"stock_qty": stock_qty, "updated_at": utcnow()}})
    return serialize(db, db.seller_products.find_one({"_id": old["_id"]}), owner=True)


# ---------- public side ----------

def list_public(db, q, category, subcategory, sort, page, page_size) -> dict:
    flt = public_filter(db)
    if q and q.strip():
        flt["$or"] = [{"title": _rx(q)}, {"brand": _rx(q)}]
    if category:
        flt["category_slug"] = category
    if subcategory:
        flt["subcategory_slug"] = subcategory
    order = {"price_asc": [("price_paisa", 1), ("_id", 1)], "price_desc": [("price_paisa", -1), ("_id", 1)],
             "newest": [("created_at", -1), ("_id", 1)]}[sort]
    total = db.seller_products.count_documents(flt)
    docs = list(db.seller_products.find(flt).sort(order).skip((page - 1) * page_size).limit(page_size))
    names = business_names(db, [d["seller_id"] for d in docs])
    return _page([serialize(db, d, names) for d in docs], total, page, page_size)


def get_public(db, product_id: str) -> dict:
    doc = db.seller_products.find_one({"_id": to_object_id(product_id, "Product")})
    if not doc:
        raise AppError(404, "NOT_FOUND", "Product not found.")
    visible = db.seller_products.find_one({**public_filter(db), "_id": doc["_id"]})
    if not visible:
        raise AppError(404, "NOT_FOUND", "Product not found.")
    return serialize(db, visible)


==================== FILE: backend/app/services/time_utils.py ====================

"""Small time helpers. MongoDB returns naive UTC datetimes, so everything is made timezone-aware here."""
from datetime import datetime, timezone


def utcnow() -> datetime:
    return datetime.now(timezone.utc)


def aware(dt):
    if dt is None:
        return None
    return dt if dt.tzinfo else dt.replace(tzinfo=timezone.utc)


def iso(dt):
    """ISO string with an explicit UTC offset, so the browser never reads it as local time."""
    a = aware(dt)
    return a.isoformat() if a else None

from fastapi import APIRouter, Depends, Query, Response

from app.core.deps import require_approved_seller, require_role
from app.db.mongo import get_db
from app.models.seller_product import (InventoryUpdate, OwnerProductPage, SellerProductCreate, SellerProductOwner,
                                       SellerProductPage, SellerProductPublic)
from app.services import product_service

router = APIRouter(tags=["Seller products"])


# ---------- public (approved sellers, active listings only) ----------
@router.get("/seller-products", response_model=SellerProductPage)
def browse(
    q: str = Query("", max_length=100),
    category: Optional[str] = None,
    subcategory: Optional[str] = None,
    sort: Literal["price_asc", "price_desc", "newest"] = "newest",
    page: int = Query(1, ge=1),
    page_size: int = Query(20, ge=1, le=50),
    db=Depends(get_db),
):
    return product_service.list_public(db, q, category, subcategory, sort, page, page_size)


@router.get("/seller-products/{product_id}", response_model=SellerProductPublic)
def detail(product_id: str, db=Depends(get_db)):
    return product_service.get_public(db, product_id)


# ---------- seller (owner only) ----------
@router.post("/seller/products", response_model=SellerProductOwner, status_code=201)
def create(body: SellerProductCreate, seller=Depends(require_approved_seller), db=Depends(get_db)):
    return product_service.create_product(db, seller, body)


@router.get("/seller/products", response_model=OwnerProductPage)
def my_products(
    page: int = Query(1, ge=1),
    page_size: int = Query(20, ge=1, le=50),
    seller=Depends(require_role("seller")),
    db=Depends(get_db),
):
    return product_service.list_own(db, seller, page, page_size)


@router.put("/seller/products/{product_id}", response_model=SellerProductOwner)
def update(product_id: str, body: SellerProductCreate, seller=Depends(require_approved_seller), db=Depends(get_db)):
    return product_service.update_product(db, seller, product_id, body)


@router.delete("/seller/products/{product_id}", status_code=204)
def remove(product_id: str, seller=Depends(require_approved_seller), db=Depends(get_db)):
    product_service.remove_product(db, seller, product_id)
    return Response(status_code=204)


@router.patch("/seller/products/{product_id}/inventory", response_model=SellerProductOwner)
def inventory(product_id: str, body: InventoryUpdate, seller=Depends(require_approved_seller), db=Depends(get_db)):
    return product_service.set_inventory(db, seller, product_id, body.stock_qty)

def test_effective_price_uses_active_promotions_only(client, db, make_account):
    user, h = approved_seller(make_account, db)
    pid = create_product(client, h)["id"]
    now = datetime.now(timezone.utc)
    promo = {"seller_id": ObjectId(user["id"]), "product_ids": [ObjectId(pid)], "discount_percent": 20,
             "starts_at": now - timedelta(days=1), "ends_at": now + timedelta(days=1), "is_active": True}
    db.promotions.insert_one(promo.copy())
    d = client.get(f"/api/seller-products/{pid}").json()
    assert d["price_paisa"] == 70000 and d["effective_price_paisa"] == 56000 and d["discount_percent"] == 20
    db.promotions.delete_many({})
    db.promotions.insert_one({**promo, "ends_at": now - timedelta(hours=1)})            # expired
    db.promotions.insert_one({**promo, "starts_at": now + timedelta(days=1), "ends_at": now + timedelta(days=2)})  # future
    db.promotions.insert_one({**promo, "is_active": False})                              # switched off
    assert client.get(f"/api/seller-products/{pid}").json()["effective_price_paisa"] == 70000

# Database Schema (all collections)

Conventions: `_id` ObjectId (exposed as `id`); `created_at`/`updated_at` UTC; money in integer paisa; category links use slugs.
Created by Member 1 now: users, seller_profiles, categories, subcategories. The rest are defined here so later members agree on field names.

## Member 1
**users** â€” email (unique, lowercase), full_name, phone, password_hash, role (user|seller|admin), is_active, verification_status (none|pending|approved|rejected), created_at, updated_at.
**seller_profiles** â€” user_id (unique), business_name, description, contact_phone, address, status (pending|approved|rejected|suspended), created_at, updated_at.
**categories** â€” slug (unique), name, is_active, order.
**subcategories** â€” slug (unique), category_slug (indexed), name, is_active.

## Member 2
**seller_products** â€” seller_id, title, description, brand, category_slug, subcategory_slug, price_paisa (>0), currency, stock_qty (>=0), status (active|inactive|removed), moderation_status (visible|hidden), images[], unit, attributes{}, source="fairshare". Indexes: seller_id, (category_slug, subcategory_slug), text(title, brand).
**external_products** â€” store (daraz|pickaboo|chaldal|shwapno|agora|rokomari), store_product_id, name, brand, model, price_paisa, currency, image_url, product_url (required), category_slug, subcategory_slug, specs{}, quantity, unit, package_size, availability, normalized_key, first_seen_at, last_seen_at, last_successful_at, freshness (fresh|stale|unavailable), source="external". Unique (store, store_product_id). NO stock/cart fields.
**price_observations** â€” external_product_id, price_paisa, availability, observed_at (append-only).
**product_matches** â€” group_key (unique), external_product_ids[], seller_product_ids[], match_type (exact|possible), reasons[], differences[].
**scrape_sources** â€” store (unique), status (success|partial|failure|unsupported|never_run), last_attempt_at, last_success_at, last_error, items_found, robots_allowed, notes.
**scrape_runs** â€” store, started_at, finished_at, status, items_found, errors[].

## Member 3
**promotions** (contract used by Member 2's `effective_price`) â€” seller_id, product_ids[] (seller_products ids), discount_percent (1-90), starts_at, ends_at, is_active. Effective price = price_paisa reduced by the largest active promotion covering now; computed server-side only.
**carts** â€” user_id (unique), items[{seller_product_id, quantity}]. No prices stored.
**orders** â€” user_id, order_number, status (pending|confirmed|shipped|delivered|cancelled), subtotal_paisa, total_paisa, payment{mode:"simulated", status}, shipping_address, created_at.
**order_items** â€” order_id, seller_id, seller_product_id, title_snapshot, unit_price_paisa, quantity, item_status.
**messages** â€” thread participants (customer_id, seller_id), order_id|product_id (optional), sender_id, body, read, created_at.
**rental_listings** â€” owner_id, title, description, daily_rate_paisa, deposit_paisa, location, images[], status, available_from, available_to. Uses NO shopping taxonomy.
**rental_requests** â€” listing_id, renter_id, start_date, end_date, status (pending|accepted|rejected|cancelled).
**rental_bookings** â€” request_id, listing_id, owner_id, renter_id, start_date, end_date, status (confirmed|active|completed|cancelled).
**rental_events** â€” booking_id|request_id, event, actor_id, at.
**rental_messages** â€” request_id, sender_id, body, created_at.

## Member 4
**verification_records** â€” user_id (unique), nid_encrypted (Fernet), nid_last4, doc_path (private, random filename), status, reviewed_by, reviewed_at, note. NEVER returned by ordinary APIs.
**complaints** â€” filer_id, target_type (order|seller|rental), target_id, description, status (open|in_review|resolved|dismissed), admin_note.
**audit_logs** â€” actor_id, action, target, at (admin actions and every document access).

## Sensitive data
Stored: encrypted NID, last 4 digits, one private document path. Never stored/logged: plaintext passwords, raw NID in logs. Files live in `private_uploads/` (git-ignored, never served statically).

# API Conventions

- Base path: `/api`. Swagger UI: `/docs`.
- Auth header: `Authorization: Bearer <JWT>`.
- JSON field names: `snake_case`. IDs are strings exposed as `id`.
- Money: integer paisa in fields ending `_paisa`; currency `BDT`.
- Dates: ISO 8601 UTC.
- Pagination: `?page=1&page_size=20` -> `{ "items": [...], "total": n, "page": 1, "page_size": 20 }`.

## Error format (always)
```json
{ "error": { "code": "EMAIL_EXISTS", "message": "Human readable text.", "details": null } }
```
| Status | Meaning |
|---|---|
| 400 | Bad request logic (e.g. NOT_A_FAIRSHARE_PRODUCT) |
| 401 | Not logged in / invalid token |
| 403 | Logged in but not allowed (role, ownership, approval) |
| 404 | Not found (also used to hide other users' records) |
| 409 | Conflict (duplicate email, out of stock, date overlap) |
| 422 | Validation error (`VALIDATION_ERROR` with field list, or `INVALID_CATEGORY`) |

## Foundation endpoints (Member 1)
| Method | Route | Role | Notes |
|---|---|---|---|
| GET | /api/health | public | `{status, database}` |
| POST | /api/auth/register | public | role `user` or `seller` only; seller needs `business_name` |
| POST | /api/auth/login | public | returns `{access_token, token_type, user}` |
| GET | /api/auth/me | any logged in | user + `seller_status` for sellers |
| GET | /api/categories | public | nested categories/subcategories (active only) |

## Rules for everyone
Validate server-side. Never trust client price/ownership. Use `require_role` / `require_approved_seller`.
Validate categories with `category_service.validate_category_pair`.

# Git Workflow (reference for later; agree as a team before using)

Branches: `main` (final release only), `develop` (integration), `feature/m1-foundation`, `feature/m2-scraping`, `feature/m3-commerce-rentals`, `feature/m4-admin-final`.

## Principles
- Commit only files you own, and add them by name (`git add <path>`), never everything blindly.
- Never force-push. Never reset or discard another member's changes.
- The sender opens a Pull Request into `develop`; the receiving member reviews it, runs the setup and both test suites, and only then merges.

## Handover
1. Sender pushes the feature branch and opens a PR into `develop`.
2. Receiver checks out the branch on a clean copy, follows docs/SETUP.md, runs `pytest` and `npm test`.
3. Receiver ticks docs/HANDOVER_Mx.md with the real results, merges, and branches from `develop` for their own phase.

## Conflicts
Pull `develop` into your branch, open the conflicted file, keep both members' marked blocks (`--- Member N ---`), run the tests, then commit.

// The ONE place that talks to the backend. Adds the JWT and normalizes errors.
const BASE = import.meta.env.VITE_API_URL || "http://localhost:8000/api";
const TOKEN_KEY = "fairshare_token";

export const tokenStore = {
  get: () => localStorage.getItem(TOKEN_KEY),
  set: (t) => localStorage.setItem(TOKEN_KEY, t),
  clear: () => localStorage.removeItem(TOKEN_KEY),
};

export class ApiError extends Error {
  constructor(message, status = 0, code = "ERROR", details = null) {
    super(message);
    this.name = "ApiError";
    this.status = status;
    this.code = code;
    this.details = details;
  }
}

export async function request(path, { method = "GET", body, auth = true } = {}) {
  const headers = { Accept: "application/json" };
  if (body !== undefined) headers["Content-Type"] = "application/json";
  const token = tokenStore.get();
  if (auth && token) headers.Authorization = `Bearer ${token}`;

  let res;
  try {
    res = await fetch(`${BASE}${path}`, {
      method,
      headers,
      body: body !== undefined ? JSON.stringify(body) : undefined,
    });
  } catch {
    throw new ApiError("Cannot reach the server. Check that the backend is running.", 0, "NETWORK_ERROR");
  }

  let data = null;
  if (res.status !== 204) {
    try {
      data = await res.json();
    } catch {
      data = null;
    }
  }

  if (!res.ok) {
    if (res.status === 401 && auth) tokenStore.clear();
    const err = data && data.error ? data.error : {};
    throw new ApiError(err.message || `Request failed (${res.status}).`, res.status, err.code || "ERROR", err.details || null);
  }
  return data;
}

import { request } from "./client.js";

// Only non-empty values go into the query string.
function qs(params = {}) {
  const p = new URLSearchParams();
  Object.entries(params).forEach(([k, v]) => {
    if (v !== undefined && v !== null && v !== "") p.set(k, v);
  });
  const s = p.toString();
  return s ? `?${s}` : "";
}

export const searchProducts = (params) => request(`/compare/search${qs(params)}`, { auth: false });
export const getGroup = (groupKey) => request(`/compare/groups/${encodeURIComponent(groupKey)}`, { auth: false });
export const getSources = () => request("/compare/sources", { auth: false });
export const getCategories = () => request("/categories", { auth: false });
export const browseSellerProducts = (params) => request(`/seller-products${qs(params)}`, { auth: false });
export const getSellerProduct = (id) => request(`/seller-products/${id}`, { auth: false });

import { createContext, useCallback, useContext, useEffect, useMemo, useState } from "react";
import { loginRequest, meRequest, registerRequest } from "../api/auth.js";
import { ApiError, tokenStore } from "../api/client.js";

export const AuthContext = createContext(null);

export function useAuth() {
  const ctx = useContext(AuthContext);
  if (!ctx) throw new Error("useAuth must be used inside AuthProvider");
  return ctx;
}

export function AuthProvider({ children }) {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    let cancelled = false;
    async function restore() {
      if (!tokenStore.get()) {
        setLoading(false);
        return;
      }
      try {
        const me = await meRequest();
        if (!cancelled) setUser(me);
      } catch {
        tokenStore.clear();
      } finally {
        if (!cancelled) setLoading(false);
      }
    }
    restore();
    return () => {
      cancelled = true;
    };
  }, []);

  // expectedRole stops e.g. a shopper from signing in on the seller/admin login page.
  const login = useCallback(async (email, password, expectedRole) => {
    const data = await loginRequest(email, password);
    if (expectedRole && data.user.role !== expectedRole) {
      throw new ApiError(`This is not a ${expectedRole} account. Use the correct login page.`, 403, "WRONG_ROLE");
    }
    tokenStore.set(data.access_token);
    setUser(data.user);
    return data.user;
  }, []);

  const register = useCallback(
    async (payload) => {
      await registerRequest(payload);
      return login(payload.email, payload.password, payload.role);
    },
    [login]
  );

  const logout = useCallback(() => {
    tokenStore.clear();
    setUser(null);
  }, []);

  const value = useMemo(() => ({ user, loading, login, register, logout }), [user, loading, login, register, logout]);
  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

// Display helpers. Money arrives as integer paisa (BDT x 100); missing data is shown as "Not available", never invented.
export function formatPrice(paisa, currency) {
  if (paisa === null || paisa === undefined) return "Not available";
  const amount = (paisa / 100).toLocaleString("en-BD", {
    minimumFractionDigits: paisa % 100 ? 2 : 0,
    maximumFractionDigits: 2,
  });
  return currency === "BDT" ? `à§³ ${amount}` : `${amount} (currency not stated)`;
}

export function formatSize(quantity, unit) {
  if (quantity === null || quantity === undefined || !unit) return null;
  if (unit === "g" && quantity >= 1000) return `${quantity / 1000} kg`;
  if (unit === "ml" && quantity >= 1000) return `${quantity / 1000} L`;
  return `${quantity} ${unit}`;
}

export function formatDate(iso) {
  if (!iso) return "Not available";
  const d = new Date(iso);
  return Number.isNaN(d.getTime()) ? "Not available" : d.toLocaleString();
}
